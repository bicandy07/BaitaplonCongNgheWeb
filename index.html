Trang Debug

1. Mã nguồn lỗi nguyên văn:
let btn = document.getElementById("23010001_btn_save");
btn.addEventListener("click", saveDate());
function saveDate() {
console.log("Dữ liệu đã được lưu lúc: " + new Date());
alert("Thành công")
}
2. Đoạn mã đã sửa lỗi hoàn chỉnh: 
// Bước 1: Đảm bảo có nút button này trong HTML để tránh lỗi Null
// <button id="23010001_btn_save" class="btn btn-danger">Lưu Thử Nghiệm</button>

let btn = document.getElementById("23010001_btn_save");

/* SỬA LỖI 1: Thay vì truyền vào "saveDate()" (gọi hàm ngay lập tức khi tải trang),
  ta chỉ truyền TÊN HÀM "saveDate" làm tham chiếu (callback) để nó chỉ chạy khi click.
*/
if(btn) { // Kiểm tra để tránh lỗi nếu không tìm thấy phần tử
    btn.addEventListener("click", saveDate); 
}

function saveDate() {
    console.log("Dữ liệu đã được lưu lúc: " + new Date());
    alert("Thành công"); // SỬA LỖI 2: Thêm dấu chấm phẩy (;) theo chuẩn convention sạch
}
